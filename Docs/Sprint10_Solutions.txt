MEDIUM
CraftingGrid.cs:

    // How many recipes could be crafted right now from what is in the inventory.
    public int CountRecipesYouCanMake()
    {
        int count = 0;

        for (int i = 0; i < recipes.Count; i++)
        {
            if (InventoryHasIngredients(recipes[i]))
            {
                count++;
            }
        }

        return count;
    }

    // True when the inventory holds enough of every ingredient of this recipe.
    private bool InventoryHasIngredients(CraftingRecipe recipe)
    {
        for (int i = 0; i < recipe.Ingredients.Count; i++)
        {
            CraftingIngredient ingredient = recipe.Ingredients[i];

            if (CountInInventory(ingredient.item) < ingredient.count)
            {
                return false;
            }
        }

        return true;
    }

    // Adds up this item over every inventory slot.
    private int CountInInventory(ItemData item)
    {
        int count = 0;

        for (int i = 0; i < inventory.SlotCount; i++)
        {
            InventorySlot slot = inventory.GetSlot(i);
            if (slot.item == item)
            {
                count += slot.count;
            }
        }

        return count;
    }
